## Derive addresses from a symbol

Every PDA above comes from the market symbol. The mints aren't PDAs, so pass them in from the table above or read them from the Market account (`longMint`, `shortMint`). The same goes for `collateralVault` and `feeRecipient`.

```typescript
import { getAssociatedTokenAddressSync } from "@solana/spl-token";
import { PublicKey } from "@solana/web3.js";

const MINT_REDEEM = new PublicKey("5MBjhNUUguLTPNR5WG6YBUUw7vUcxQ14ARw3NsS3rKu4");
const CLP = new PublicKey("8xauDRjw9XRyk4FE3hW1JKjD8nC87gfr59Xig1dJqLES");
const ORACLE = new PublicKey("5vxiCrDpFnQ2W5QtgZBC66K2XTC19bjVBjinGYYBsadC");
const CUSDC = new PublicKey("B1c5xBYkp7AAemYhcu4VuH4CU4sPJDDuG2iuv6ts38uE");

const pda = (seeds: Buffer[], programId: PublicKey) =>
  PublicKey.findProgramAddressSync(seeds, programId)[0];

export function marketAddresses(symbol: string, longMint: PublicKey, shortMint: PublicKey) {
  const market = pda([Buffer.from("market"), Buffer.from(symbol)], MINT_REDEEM);
  const clp = pda([Buffer.from("clp"), market.toBuffer()], CLP);
  const pool = (side: number, synthMint: PublicKey) => {
    const address = pda([Buffer.from("pool"), market.toBuffer(), Buffer.from([side])], CLP);
    return {
      address,
      synthVault: getAssociatedTokenAddressSync(synthMint, address, true),
      usdcVault: getAssociatedTokenAddressSync(CUSDC, address, true),
    };
  };
  return {
    market,
    oracle: pda([Buffer.from("oracle_config"), market.toBuffer()], ORACLE),
    clp,
    clpVault: getAssociatedTokenAddressSync(CUSDC, clp, true),
    longPool: pool(0, longMint),
    shortPool: pool(1, shortMint),
  };
}

// marketAddresses("XAU", xauLongMint, xauShortMint).longPool.address
//   => 5rEH7sYUS7kuqNiLZmCiQGwQLmQ1qqogRPavT9csKmHL
```
